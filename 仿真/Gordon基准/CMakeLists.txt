cmake_minimum_required(VERSION 3.16)
project(filter_baselines VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_library(baseline
  src/models/model.cpp src/benchmark/io.cpp src/benchmark/generator.cpp
  src/filters/statistics.cpp src/filters/gaussian.cpp src/filters/particles.cpp
  src/ot/sinkhorn.cpp src/evaluation/evaluation.cpp src/evaluation/time_series.cpp)
target_include_directories(baseline PUBLIC include)
target_link_libraries(baseline PUBLIC Eigen3::Eigen)
# 禁用内部线程，任务并行由调度器控制，保留 IEEE 浮点语义
target_compile_definitions(baseline PUBLIC EIGEN_DONT_PARALLELIZE)
target_compile_options(baseline PRIVATE -Wall -Wextra -Wpedantic)
add_executable(generate apps/generate.cpp)
add_executable(filter apps/filter.cpp)
add_executable(evaluate apps/evaluate.cpp)
add_executable(summarize apps/summarize.cpp)
add_executable(time_series apps/time_series.cpp)
target_link_libraries(generate PRIVATE baseline)
target_link_libraries(filter PRIVATE baseline)
target_link_libraries(evaluate PRIVATE baseline)
target_link_libraries(summarize PRIVATE baseline)
target_link_libraries(time_series PRIVATE baseline)
enable_testing()
add_executable(test_baseline tests/test_baseline.cpp)
target_link_libraries(test_baseline PRIVATE baseline)
add_test(NAME baseline_math COMMAND test_baseline)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME pipeline_contract COMMAND Python3::Interpreter tests/test_pipeline.py)
set_tests_properties(pipeline_contract PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
# 网格参考独立于四个基线算法，仅用于一维后验积分精度核对。
add_library(grid_reference_core src/reference/grid_reference.cpp)
target_link_libraries(grid_reference_core PUBLIC baseline)
target_compile_options(grid_reference_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(grid_reference apps/grid_reference.cpp)
target_link_libraries(grid_reference PRIVATE grid_reference_core)
add_executable(grid_risk apps/grid_risk.cpp)
target_link_libraries(grid_risk PRIVATE baseline)
add_executable(test_grid_reference tests/test_grid_reference.cpp)
target_link_libraries(test_grid_reference PRIVATE grid_reference_core)
add_test(NAME grid_reference_math COMMAND test_grid_reference)
add_test(NAME grid_risk_math COMMAND Python3::Interpreter "${CMAKE_SOURCE_DIR}/tests/test_grid_risk.py")
add_executable(paper_statistics apps/paper_statistics.cpp)
target_link_libraries(paper_statistics PRIVATE baseline)
# 后验形状诊断单独构建，不改变已冻结滤波二进制的哈希与计时。
add_library(density_core src/evaluation/density.cpp)
target_link_libraries(density_core PUBLIC baseline)
add_executable(posterior_snapshot apps/posterior_snapshot.cpp)
target_link_libraries(posterior_snapshot PRIVATE baseline)
add_executable(kde_density apps/kde_density.cpp)
target_link_libraries(kde_density PRIVATE density_core)
add_executable(test_density tests/test_density.cpp)
target_link_libraries(test_density PRIVATE density_core)
add_test(NAME density_math COMMAND test_density)
