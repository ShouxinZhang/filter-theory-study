cmake_minimum_required(VERSION 3.20)
project(filter_benchmark VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
option(FILTER_SANITIZERS "启用地址与未定义行为检查" OFF)
add_compile_options(-Wall -Wextra -Wpedantic)
if(FILTER_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
# 依赖方向: 公共数值层 -> 环境/滤波/存储 -> 驱动, 评估不依赖滤波实现。
add_library(filter_core src/core/random.cpp src/core/numerics.cpp)
target_include_directories(filter_core PUBLIC include)
target_link_libraries(filter_core PUBLIC Eigen3::Eigen)
add_library(noise src/noise/distribution.cpp src/noise/serialization.cpp)
target_link_libraries(noise PUBLIC filter_core)
add_library(environments src/environment/gordon.cpp src/environment/tracking.cpp src/environment/noise_model.cpp src/environment/factory.cpp)
target_link_libraries(environments PUBLIC noise)
add_library(transport src/transport/monotone.cpp src/transport/network_simplex.cpp)
target_link_libraries(transport PUBLIC filter_core)
target_include_directories(transport SYSTEM PRIVATE third_party/pot)
add_library(filters src/filter/particle_filter.cpp src/filter/resampling.cpp)
target_link_libraries(filters PUBLIC environments transport)
add_library(storage src/io/binary.cpp src/io/dataset.cpp src/io/results.cpp)
target_link_libraries(storage PUBLIC filter_core OpenSSL::Crypto)
add_library(benchmark src/benchmark/generate.cpp)
target_link_libraries(benchmark PUBLIC environments storage)
add_library(reference src/reference/grid.cpp)
target_link_libraries(reference PUBLIC environments storage)
add_library(evaluation src/evaluation/metrics.cpp src/evaluation/report.cpp)
target_link_libraries(evaluation PUBLIC storage)
add_executable(filter_bench apps/main.cpp apps/options.cpp apps/noise_options.cpp apps/commands.cpp apps/run.cpp)
target_link_libraries(filter_bench PRIVATE benchmark filters reference evaluation Threads::Threads)
target_compile_definitions(filter_bench PRIVATE FILTER_BUILD_TYPE="${CMAKE_BUILD_TYPE}")
string(TOUPPER "${CMAKE_BUILD_TYPE}" FILTER_BUILD_UPPER)
target_compile_definitions(filter_bench PRIVATE FILTER_BUILD_FLAGS="${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${FILTER_BUILD_UPPER}}")
include(CTest)
if(BUILD_TESTING)
  add_executable(filter_tests tests/main.cpp tests/models.cpp tests/noise.cpp tests/transport.cpp tests/integration.cpp tests/reference.cpp)
  target_link_libraries(filter_tests PRIVATE benchmark filters reference evaluation)
  add_test(NAME mathematical_contracts COMMAND filter_tests)
  add_test(NAME cli_pipeline COMMAND ${CMAKE_COMMAND} -DBENCH=$<TARGET_FILE:filter_bench>
    -DWORK=${CMAKE_CURRENT_BINARY_DIR}/cli-test -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
  add_test(NAME noise_cli_pipeline COMMAND ${CMAKE_COMMAND} -DBENCH=$<TARGET_FILE:filter_bench>
    -DWORK=${CMAKE_CURRENT_BINARY_DIR}/noise-cli-test -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/noise_cli.cmake)
endif()
