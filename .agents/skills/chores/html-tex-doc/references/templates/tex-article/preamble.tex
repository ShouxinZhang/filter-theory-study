\newcommand{\htdfig}[3]{%
  \begin{figure}[htbp]
    \centering
    \filename@parse{html/src/#1}%
    \in@{algs/}{#1}%
    \ifin@
      \begin{minipage}{\linewidth}%
        \includestandalone{\filename@area\filename@base}%
      \end{minipage}%
    \else
      \IfFileExists{\filename@area\filename@base.pdf}{%
        \includegraphics[width=\linewidth,height=0.72\textheight,keepaspectratio]{\filename@area\filename@base.pdf}%
      }{%
        \begin{lrbox}{\htd@figbox}%
          \includestandalone{\filename@area\filename@base}%
        \end{lrbox}%
        \ifdim\wd\htd@figbox>\linewidth
          \resizebox{\linewidth}{!}{\usebox{\htd@figbox}}%
        \else
          \usebox{\htd@figbox}%
        \fi
      }%
    \fi
    \caption{#3}\label{#2}%
  \end{figure}%
}
